13️⃣ Mini projekt – Kalkulačka


Výsledok: ?

💻 Ukážka kódu

<div class="kalkulacka">
  <input type="number" id="cislo1" placeholder="Číslo 1">
  <select id="operacia">
    <option value="+">+</option>
    <option value="-">-</option>
    <option value="*">×</option>
    <option value="/">÷</option>
  </select>
  <input type="number" id="cislo2" placeholder="Číslo 2">
  <button onclick="vypocitaj()">Vypočítať</button>
  <div id="vysledok">Výsledok: ?</div>
</div>

<script>
function vypocitaj() {
  const a = parseFloat(document.getElementById('cislo1').value);
  const b = parseFloat(document.getElementById('cislo2').value);
  const op = document.getElementById('operacia').value;
  let vysledok;
  if(op === '+') vysledok = a + b;
  else if(op === '-') vysledok = a - b;
  else if(op === '*') vysledok = a * b;
  else if(op === '/') vysledok = b !== 0 ? a / b : 'Chyba';
  document.getElementById('vysledok').innerText = 'Výsledok: ' + vysledok;
}
</script>
⬅️ Späť na hlavnú stránku
